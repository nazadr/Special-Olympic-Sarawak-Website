-- ============================================
-- PARTICIPANTS MANAGEMENT DATABASE STRUCTURE
-- Athletes, Coaches & Volunteers Tables
-- Special Olympics Sarawak
-- ============================================

-- Athletes Table
CREATE TABLE IF NOT EXISTS athletes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    excel_row_id INT NULL COMMENT 'Row number from Excel for reference',
    full_name VARCHAR(255) NOT NULL,
    email VARCHAR(255) NULL,
    phone VARCHAR(50) NULL,
    date_of_birth DATE NULL,
    gender ENUM('Male', 'Female', 'Other') NULL,
    chapter VARCHAR(100) NULL COMMENT 'Kuching, Miri, Sibu, Bintulu, Samarahan',
    sports_interested TEXT NULL COMMENT 'Sports they want to participate in',
    medical_conditions TEXT NULL,
    emergency_contact_name VARCHAR(255) NULL,
    emergency_contact_phone VARCHAR(50) NULL,
    registration_date DATETIME NULL,
    status ENUM('pending', 'approved', 'rejected', 'archived') DEFAULT 'pending',
    notes TEXT NULL,
    excel_filename VARCHAR(255) NULL COMMENT 'Source Excel file name',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_status (status),
    INDEX idx_chapter (chapter),
    INDEX idx_created_at (created_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Coaches Table
CREATE TABLE IF NOT EXISTS coaches (
    id INT AUTO_INCREMENT PRIMARY KEY,
    excel_row_id INT NULL,
    full_name VARCHAR(255) NOT NULL,
    email VARCHAR(255) NULL,
    phone VARCHAR(50) NULL,
    date_of_birth DATE NULL,
    gender ENUM('Male', 'Female', 'Other') NULL,
    chapter VARCHAR(100) NULL,
    sports_expertise TEXT NULL COMMENT 'Sports they can coach',
    experience_years INT NULL,
    certifications TEXT NULL COMMENT 'Coaching certifications',
    availability VARCHAR(255) NULL COMMENT 'Days/times available',
    emergency_contact_name VARCHAR(255) NULL,
    emergency_contact_phone VARCHAR(50) NULL,
    registration_date DATETIME NULL,
    status ENUM('pending', 'approved', 'rejected', 'archived') DEFAULT 'pending',
    notes TEXT NULL,
    excel_filename VARCHAR(255) NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_status (status),
    INDEX idx_chapter (chapter),
    INDEX idx_created_at (created_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Volunteers Table
CREATE TABLE IF NOT EXISTS volunteers (
    id INT AUTO_INCREMENT PRIMARY KEY,
    excel_row_id INT NULL,
    full_name VARCHAR(255) NOT NULL,
    email VARCHAR(255) NULL,
    phone VARCHAR(50) NULL,
    date_of_birth DATE NULL,
    gender ENUM('Male', 'Female', 'Other') NULL,
    chapter VARCHAR(100) NULL,
    volunteer_role TEXT NULL COMMENT 'Preferred volunteer roles',
    skills TEXT NULL COMMENT 'Skills they can offer',
    availability VARCHAR(255) NULL,
    previous_volunteer_experience TEXT NULL,
    emergency_contact_name VARCHAR(255) NULL,
    emergency_contact_phone VARCHAR(50) NULL,
    registration_date DATETIME NULL,
    status ENUM('pending', 'approved', 'rejected', 'archived') DEFAULT 'pending',
    notes TEXT NULL,
    excel_filename VARCHAR(255) NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_status (status),
    INDEX idx_chapter (chapter),
    INDEX idx_created_at (created_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Excel Upload History Table (track all uploads)
CREATE TABLE IF NOT EXISTS participant_excel_uploads (
    id INT AUTO_INCREMENT PRIMARY KEY,
    participant_type ENUM('athlete', 'coach', 'volunteer') NOT NULL,
    filename VARCHAR(255) NOT NULL,
    original_filename VARCHAR(255) NOT NULL,
    file_path VARCHAR(500) NOT NULL,
    rows_imported INT DEFAULT 0,
    rows_updated INT DEFAULT 0,
    rows_failed INT DEFAULT 0,
    uploaded_by INT NULL COMMENT 'admin_id',
    upload_status ENUM('success', 'partial', 'failed') DEFAULT 'success',
    error_log TEXT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_type (participant_type),
    INDEX idx_created_at (created_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ============================================
-- USEFUL QUERIES
-- ============================================

-- Get statistics for each participant type
SELECT 
    COUNT(*) as total,
    SUM(CASE WHEN status = 'approved' THEN 1 ELSE 0 END) as approved,
    SUM(CASE WHEN status = 'pending' THEN 1 ELSE 0 END) as pending,
    SUM(CASE WHEN status = 'rejected' THEN 1 ELSE 0 END) as rejected
FROM athletes;

-- Get chapter breakdown
SELECT chapter, COUNT(*) as count 
FROM athletes 
WHERE status = 'approved' 
GROUP BY chapter 
ORDER BY count DESC;

-- Get recent uploads
SELECT * FROM participant_excel_uploads 
ORDER BY created_at DESC 
LIMIT 10;
